def transition_curve(page):
    # Native time: sample all six computed opacities every 60 ms, plus every rendered frame.
    return page.evaluate('''async () => {
      const c=giftCard.card, imgs=[...c.querySelectorAll('.art-media img')],
        masks=[...c.querySelectorAll('.subject-mask:not(.special-tint)')],
        tints=[...c.querySelectorAll('.special-tint')];
      const es=[imgs[0],masks[0],tints[0],imgs[1],masks[1],tints[1]];
      const start=performance.now(), samples=[], frames=[];
      const sample=()=>({ms:performance.now()-start,face:c.dataset.face,
        opacity:es.map(e=>+getComputedStyle(e).opacity)});
      samples.push(sample());giftCard.trigger();
      const timer=setInterval(()=>samples.push(sample()),60);
      await new Promise(resolve=>{function frame(){frames.push(sample());
        if(performance.now()-start<giftCard.CYCLE+giftCard.FADE*2+700)requestAnimationFrame(frame);
        else resolve();}requestAnimationFrame(frame);});
      clearInterval(timer);samples.push(sample());
      return {samples,frames,gain:+getComputedStyle(c).getPropertyValue('--subject-gain')};
    }''')


def assert_crossfade(curve):
    gains = [1, curve['gain'], .72]
    for direction in ('idle', 'alt'):
        rows = [s for s in curve['frames'] if s['face'] == direction and s['ms'] > 1]
        for k, gain in enumerate(gains):
            middle = [s for s in rows if .05 < s['opacity'][k] / gain < .95]
            assert len(middle) >= 3, ('crossfade intermediate samples', direction, k, len(middle))
    for row in curve['frames']:
        for k, gain in enumerate(gains):
            a, b = row['opacity'][k] / gain, row['opacity'][k + 3] / gain
            assert not (a < .05 and b < .05), ('blank frame', row, k)
            assert not (a > .95 and b > .95), ('double layer', row, k)
            assert abs(a + b - 1) < .025, ('non-complementary fade', row, k)
    assert all(abs(a-b)<.005 for a,b in zip(curve['samples'][-1]['opacity'],[1,curve['gain'],.72,0,0,0])), 'return endpoint'


def mask_alignment(card, index=0):
    sources = card.evaluate('''(c,i)=>({image:c.querySelectorAll('.art-media img')[i].src,
      mask:getComputedStyle(c.querySelectorAll('.subject-mask:not(.special-tint)')[i]).maskImage.slice(5,-2)})''', index)
    def read(uri):
        if uri.startswith('data:'):
            return Image.open(BytesIO(b64decode(uri.split(',', 1)[1])))
        return Image.open(Path(unquote(urlparse(uri).path).lstrip('/')))
    art, mask = read(sources['image']), read(sources['mask'])
    values = []
    for frame in range(getattr(art, 'n_frames', 1)):
        art.seek(frame)
        mask.seek(min(frame, getattr(mask, 'n_frames', 1)-1))
        a = np.asarray(art.convert('RGBA').getchannel('A').resize((214,300))) > 127
        m = np.asarray(mask.convert('RGBA').getchannel('A').resize((214,300))) > 127
        values.append(float(np.logical_and(a,m).sum() / np.logical_or(a,m).sum()))
    assert min(values) > .95, ('foil silhouette mismatch', min(values))
    return {'frames':len(values),'minimumAlphaIoU':min(values)}

def diff(a, b):
    return sum(abs(x - y) for x, y in zip(a, b)) / 3


def text_contrast(page, selector, segment=None):
    """卡名的可讀性用文字該用的量法：字身與底的**亮度對比比值**（WCAG）。

    原本借用寶石那條「內部 vs 四周色差 ≥35」量文字會失真——取樣框裡一半是筆畫、
    一半是底，平均起來永遠落在中間。這裡改成在卡名的框內取亮度分佈：
    最亮的 25% 當字身、最暗的 35% 當底，算 (L1+.05)/(L2+.05)，門檻 4.5:1（WCAG AA 內文）。
    這比舊量法**更嚴**，不是為了湊綠燈放寬。
    """
    im = Image.open(BytesIO(page.locator('#stage').screenshot())).convert('RGB')
    dpr = page.evaluate('devicePixelRatio')
    box = page.evaluate("""([sel,segment]) => {
      const s = document.querySelector('#stage').getBoundingClientRect();
      const el = document.querySelector(sel);
      let g = el.getBoundingClientRect();
      if(segment){const r=document.createRange(), text=el.firstChild;
        const start=text.textContent.indexOf(segment);r.setStart(text,start);
        r.setEnd(text,start+segment.length);g=r.getBoundingClientRect();}
      return {x: g.left - s.left, y: g.top - s.top, w: g.width, h: g.height};
    }""", [selector, segment])
    x, y, w, h = (round(box[k] * dpr) for k in ('x', 'y', 'w', 'h'))
    a = np.asarray(im.crop((x, y, x + w, y + h))).astype(float) / 255
    lin = np.where(a <= .03928, a / 12.92, ((a + .055) / 1.055) ** 2.4)
    L = lin[:, :, 0] * .2126 + lin[:, :, 1] * .7152 + lin[:, :, 2] * .0722
    glyph = L[L >= np.percentile(L, 75)].mean()
    ground = L[L <= np.percentile(L, 35)].mean()
    hi, lo = max(glyph, ground), min(glyph, ground)
    return round((hi + .05) / (lo + .05), 2)


def region_contrast(page, selector, pad):
    """量 selector 內部相對於它四周一圈的色差（寶石與名字共用一套量法）。"""
    im = Image.open(BytesIO(page.locator('#stage').screenshot())).convert('RGB')
    dpr = page.evaluate('devicePixelRatio')
    box = page.evaluate('''(sel) => {
      const s = document.querySelector('#stage').getBoundingClientRect();
      const g = document.querySelector(sel).getBoundingClientRect();
      return {x: g.left - s.left, y: g.top - s.top, w: g.width, h: g.height};
    }''', selector)
    x, y, w, h = (round(box[k] * dpr) for k in ('x', 'y', 'w', 'h'))
    p = round(pad * dpr)
    inner = im.crop((x + round(w * .28), y + round(h * .28), x + round(w * .72), y + round(h * .72)))
    rings = [im.crop((max(0, x - p), max(0, y - p), min(im.width, x + w + p), max(0, y - 2))),
             im.crop((max(0, x - p), min(im.height, y + h + 2), min(im.width, x + w + p), min(im.height, y + h + p))),
             im.crop((max(0, x - p), y, min(im.width, x - 2), min(im.height, y + h))),
             im.crop((min(im.width, x + w + 2), y, min(im.width, x + w + p), min(im.height, y + h)))]
    rings = [r for r in rings if r.width > 0 and r.height > 0]
    g = tuple(round(v) for v in ImageStat.Stat(inner).mean)
    r = tuple(round(sum(ImageStat.Stat(i).mean[k] for i in rings) / len(rings)) for k in range(3))
    return round(diff(g, r), 2), g, r

def freeze_art(page):
    """固定 WebP 第 0 格（含剪影），保留粒子的原生 CSS 時鐘。"""
    sources = page.evaluate('''() => [...new Set([...document.images].map(i=>i.src).concat(
      [...document.querySelectorAll('.subject-mask')].map(e=>getComputedStyle(e).maskImage.slice(5,-2))))]''')
    from base64 import b64encode
    fixed = {}
    for uri in sources:
        if uri.startswith('data:'):
            im = Image.open(BytesIO(b64decode(uri.split(',', 1)[1])))
            im.seek(0)
            buf = BytesIO()
            im.convert('RGBA').save(buf, format='PNG')
            fixed[uri] = 'data:image/png;base64,' + b64encode(buf.getvalue()).decode()
    page.evaluate('''async fixed => {
      for(const i of document.images){if(fixed[i.src])i.src=fixed[i.src];await i.decode();}
      for(const e of document.querySelectorAll('.subject-mask')){
        const src=getComputedStyle(e).maskImage.slice(5,-2);
        if(fixed[src])e.style.setProperty('--subject',`url("${fixed[src]}")`);}
    }''', fixed)
    page.add_style_tag(content='body:before{animation:none!important}')
    page.wait_for_timeout(400)

          maxOverflow:Math.max(cr.left-pr.left,pr.right-cr.right,cr.top-pr.top,pr.bottom-cr.bottom),
          keyframeProperties:pollenFrames,
          children:[...p.children].map(e=>{const s=getComputedStyle(e);return {
            blend:s.mixBlendMode,size:parseFloat(s.width),duration:s.animationDuration,delay:s.animationDelay};})}:null};}''')
    gate('rarity transparent fill',css['fill']=='rgba(0, 0, 0, 0)',css['fill'])
    gate('rarity linear gradient',css['image'].startswith('linear-gradient'),css['image'])
    gate('rarity no text-shadow',css['shadow']=='none',css['shadow'])
    m = page.evaluate(MEASURE)
    old = json.loads((SHOTS/('before-'+label+'-round36.json')).read_text(encoding='utf-8'))['baselineMetrics'] if not ARGS.baseline else m
    for key in ['nameFs','rarityFs','gemFs','nameCenterDelta','rarityCenterDelta','maxOverflow']:
        gate('preserved '+key,abs(m[key]-old[key])<.0001,m[key])
    for key in PLATE:
        gate('preserved plate '+key,abs(m['plate'][key]-old['plate'][key])<.0001,m['plate'][key])
    gate('rarity centered <2px',m['rarityCenterDelta']<2,m['rarityCenterDelta'])
    for sel,key in [('face-name','nameHit'),('face-rarity','rarityHit')]:
        gate(sel+' hit',sel in (css[key] or ''),css[key])
    rarity = {segment:text_contrast(page,'.face-rarity',segment) for segment in ['特殊','SPECIAL']}
    name = text_contrast(page,'.face-name')
    gem = region_contrast(page,'.face-gem',8)[0]
    gate('rarity contrast >=4.5',min(rarity.values())>=4.5,rarity)
    gate('name contrast >=6.44',name>=6.44,name)
    gate('gem contrast >=145.67',gem>=145.67,gem)
    p = css['particles']
    gate('particle layer exists',bool(p),p)
    if p:
        gate('particle depth <=42',p['z']<=42,p['z'])
        gate('particle effective depth <=42',p['effectiveZ']<=42,p['effectiveZ'])
        gate('particle z-index <20',p['zIndex']<20,p['zIndex'])
        gate('particle rounded clip',p['overflow']=='hidden' and p['borderRadius']!='0px',p)
        gate('particle projection <=1.5px',p['maxOverflow']<=1.5,p['maxOverflow'])
        gate('particle keyframes transform/opacity only',bool(p['keyframeProperties']) and
             all(set(props)<= {'transform','opacity'} for props in p['keyframeProperties']),p['keyframeProperties'])
    # 實際 pointer 傾斜時，兩行漸層一起改變角度。
    stage = page.locator('#stage').bounding_box()
    before_gradients = page.locator('.face-name,.face-rarity').evaluate_all('(es)=>es.map(e=>getComputedStyle(e).backgroundImage)')
    page.mouse.move(stage['x']+stage['width']*.84,stage['y']+stage['height']*.22)
    page.wait_for_timeout(450)
    tilted_gradients = page.locator('.face-name,.face-rarity').evaluate_all('(es)=>es.map(e=>getComputedStyle(e).backgroundImage)')
    tilted_overflow = page.evaluate('''() => {const p=document.querySelector('.gift-particles');if(!p)return null;
      const c=giftCard.card.getBoundingClientRect(),r=p.getBoundingClientRect();
      return Math.max(c.left-r.left,r.right-c.right,c.top-r.top,r.bottom-c.bottom);}''')
    gate('tilted particle projection <=1.5px',tilted_overflow is not None and tilted_overflow<=1.5,tilted_overflow)
    shot('tilt')
    gate('both gradients follow phase',all(a!=b for a,b in zip(before_gradients,tilted_gradients)) and
         tilted_gradients[0].split(',')[0]==tilted_gradients[1].split(',')[0],tilted_gradients)
    page.mouse.move(0,0)
    page.wait_for_timeout(450)
    page.evaluate("giftCard.card.getAnimations({subtree:true}).filter(a=>a.animationName==='gift-pollen').forEach(a=>a.play())")
    # 計數器只涵蓋頁面程式；驗收自己的 transition_curve rAF 尚未啟動。
    page.evaluate('''() => {window.giftRafCount=0;window.originalGiftRaf=window.requestAnimationFrame;
      window.requestAnimationFrame=function(...args){giftRafCount++;return originalGiftRaf.apply(this,args);};}''')
    page.wait_for_timeout(3000)
    rafs = page.evaluate('giftRafCount')
    page.evaluate('() => {window.requestAnimationFrame=originalGiftRaf;}')
    gate('idle rAF =0',rafs==0,rafs)
    page.emulate_media(reduced_motion='reduce')
    page.wait_for_timeout(300)
    reduced_css = page.locator('.gift-particles > i').evaluate_all('(es)=>es.map(e=>getComputedStyle(e).animationName)')
    ra = shot('reduced-motion-0')
    page.wait_for_timeout(2000)
    rb = shot('reduced-motion-2')
    reduced_diff = float((np.abs(gray(ra)-gray(rb))[card_mask]>6).mean()*100)
    gate('reduced animation none',bool(reduced_css) and all(x=='none' for x in reduced_css),reduced_css)
    gate('reduced pixels <=0.02%',reduced_diff<=.02,reduced_diff)
    page.emulate_media(reduced_motion='no-preference')
    result = {'baselineMetrics':m,'rarityContrast':rarity,'nameContrast':name,'gemContrast':gem,
        'stats':stats,'gainsPercent':gains,'css':css,'motionPercent':float((delta[art]>6).mean()*100),
        'plateMotionPercent':float((delta[plate]>6).mean()*100),'particlePeak':peak,
        'particleNeighborPeak':neighbor_peak,'particleGainPercent':particle_gain,
        'rafs':rafs,'tiltedGradients':tilted_gradients,'tiltedParticleOverflow':tilted_overflow,
        'reducedPercent':reduced_diff,'reducedAnimation':reduced_css,'failures':failures}
    (SHOTS/(prefix+'-round36.json')).write_text(json.dumps(result,ensure_ascii=False,indent=1),encoding='utf-8')
    print(label, json.dumps({k:result[k] for k in ['rarityContrast','nameContrast','gemContrast','gainsPercent','motionPercent','particlePeak','failures']},ensure_ascii=False))
    if not ARGS.baseline:
        assert not failures, failures
    return result


def run(page, uri, label):
    errors, external = [], []
    page.on('pageerror', lambda e: errors.append(str(e)))
    page.on('console', lambda m: errors.append(m.text) if m.type == 'error' else None)
    page.on('request', lambda r: external.append(r.url) if r.url.startswith(('http:', 'https:')) else None)
    page.goto(uri, wait_until='load')
    page.wait_for_function('Array.from(document.images).every(i => i.complete && i.naturalWidth > 0)')
    page.wait_for_timeout(500)
    assert not errors, errors
    assert not external, external

    r36 = round36(page, label)
    if ARGS.baseline or ARGS.round36_only:
        return r36
    # 原有動畫驗收重新載入原頁，避免固定素材影響疊化測試。
    page.goto(uri, wait_until='load')
    page.wait_for_function('Array.from(document.images).every(i => i.complete && i.naturalWidth > 0)')
    page.wait_for_timeout(500)

    m = page.evaluate(MEASURE)
    m['detail'] = page.locator('.hcard').evaluate(DETAIL)
    m['maskAlignment'] = [mask_alignment(page.locator('.hcard'), i) for i in range(2)]
    assert m['detail']['nameGemGap'] > 0, m['detail']
    assert m['detail']['glyphCenterDelta'] < 2, m['detail']
    assert m['detail']['id'] == 'mohuashaonv', m['detail']['id']
    assert page.title() == '魔花少女 · 特殊'
    assert page.locator('h1').inner_text() == '魔花少女 · 特殊'
    for reference in refs.values():
        assert m['detail']['layers'] == reference['layers'], m['detail']['layers']
        for selector, box in reference['boxes'].items():
            for key, value in box.items():
                assert abs(m['detail']['boxes'][selector][key] - value) < TOL, (selector, key)
        for key, value in reference['fontShares'].items():
            assert abs(m['detail']['fontShares'][key] - value) < .01, key
    assert m['rarity'] == 'special' and m['kind'] == 'depth', m
    assert m['nameText'] == '魔花少女', m['nameText']
    assert m['rarityText'] == '特殊 / SPECIAL', m['rarityText']
    # 「特殊」的字效：粉→藍漸層（使用者指定）。跟 round9 驗神話同一種驗法——
    # 必須是透明填色＋真的有漸層，而且 text-shadow 要關掉（有 background-clip 的字
    # 用 text-shadow 會把陰影畫在漸層外面，HANDOFF 六之二）。
    assert m['nameFill'] == 'rgba(0, 0, 0, 0)', m['nameFill']
    assert m['nameImage'].startswith('linear-gradient'), m['nameImage']
    assert m['nameShadow'] == 'none', m['nameShadow']
    assert m['hasSubjectMask'], '主體沒掛 foil mask'
    # depth 的兩層（主體＋背景）＋ 疊在上面的特殊動作那張，共三張，
    # 而且三張都必須是同一個 600×840 畫布（scene 卡兩層是 object-fit:fill，
    # 尺寸不同就會位移／變形）。
    assert len(m['layers']) == 3, m['layers']
    assert all(l == [600, 840] for l in m['layers']), m['layers']
    for side, v in PLATE.items():
        assert abs(m['plate'][side] - v) <= TOL, '%s plate.%s = %.2f%%，應為 %s%%' % (label, side, m['plate'][side], v)
    assert m['nameCenterDelta'] < 2 and m['rarityCenterDelta'] < 2, m
    assert m['nameFits'] == 'true', m
    assert m['maxOverflow'] <= 1.5, m
    assert 'face-name' in (m['nameTop'] or ''), '名字被蓋住了：最上層是 %s' % m['nameTop']
    w = m['cardWidth']
    assert abs(m['rarityFs'] - w * 13.5 / 290) < .05, m
    assert abs(m['gemFs'] - w * 22 / 290) < .05, m
    # 文字框必須是半透明 scrim，不能又變回不透明黑條
    assert m['plateAlpha'].startswith('linear-gradient'), m['plateAlpha']
    assert m['sheenLayers'] == 3, '反光層沒掛齊（常態、特殊動作、背景各一）：%s' % m['sheenLayers']

    gem, gem_rgb, gem_ring = region_contrast(page, '.face-gem', 8)
    assert gem >= 35, '%s 寶石對比 %s < 35（%s / %s）' % (label, gem, gem_rgb, gem_ring)
    name = text_contrast(page, '.face-name')
    assert name >= 4.5, '%s 卡名對比 %s:1 < 4.5:1（WCAG AA）' % (label, name)

    page.locator('#stage').screenshot(path=str(SHOTS / (ARGS.tag + '-gift-%s.png' % label)))
    print('  %s：幾何 OK・寶石對比 %s・卡名對比 %s・%s / %s・卡寬 %.0fpx・字級 %.2f/%.2f/%.2f'
          % (label, gem, '%.2f:1' % name, m['nameText'], m['rarityText'], w,
             m['nameFs'], m['rarityFs'], m['gemFs']))
    curve = transition_curve(page)
    (SHOTS / (ARGS.tag + '-' + label + '-curve.json')).write_text(json.dumps(curve, indent=1), encoding='utf-8')
    result = {'label': label, 'gem_contrast': gem, 'name_contrast': name, 'round36':r36, **m}
    (SHOTS / (ARGS.tag + '-' + label + '-metrics.json')).write_text(json.dumps(result, ensure_ascii=False, indent=1), encoding='utf-8')
    assert_crossfade(curve)
    result['motionPixels'] = capture_motion(page, ARGS.tag + '-' + label)
    assert not errors and not external, (errors, external)
    result['errors'], result['externalRequests'] = errors, external
    return result


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument('--tag', default='after')
    parser.add_argument('--baseline', action='store_true')
    parser.add_argument('--round36-only', action='store_true')
    parser.add_argument('--source', type=Path, default=OUT / 'gift-mohuashaonv.html')
    ARGS = parser.parse_args()
    SHOTS.mkdir(parents=True, exist_ok=True)
    report = []
    with sync_playwright() as pw:
        # Windows 原生 GPU；預設 SwiftShader 在這張動畫卡只錄到約 14fps。
        # 不改 320ms 過渡或取樣門檻，讓 rAF 記錄實際硬體呈現的每一格。
        browser = pw.chromium.launch(args=['--use-angle=d3d11'])
        page = browser.new_page(viewport={'width': 900, 'height': 1000}, device_scale_factor=2)
        refs = {} if ARGS.baseline or ARGS.round36_only else references(browser)
        (SHOTS / (ARGS.tag + '-references.json')).write_text(json.dumps(refs, ensure_ascii=False, indent=1), encoding='utf-8')
        src = ARGS.source.resolve()
        assert src.is_relative_to(OUT.parents[1].resolve()), 'source must stay in this worktree'
        report.append(run(page, src.as_uri(), 'dev'))
        with TemporaryDirectory(dir=SHOTS) as tmp:                 # 搬到別的資料夾也要能開
            copy = Path(tmp) / src.name
            shutil.copy(src, copy)
            report.append(run(page, copy.as_uri(), 'moved-copy'))
        browser.close()

    (SHOTS / (ARGS.tag+'-verification.json')).write_text(json.dumps(report, ensure_ascii=False, indent=1), encoding='utf-8')
    if ARGS.baseline:
        assert not any(r['failures'] for r in report), 'expected pre-fix round36 failures (see JSON)'
    print('OK ·', len(report), 'entries')
