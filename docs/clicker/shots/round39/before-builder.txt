輸出：gift-mohuashaonv.html（自包含，複製到任何資料夾都能開）

CARD = (600, 840)          # 場景卡的畫布，兩層都用這個尺寸
SUBJ_WIDTH = 0.98          # 角色寬度佔卡寬。1.2 倍會超過卡寬、翅膀被裁掉（使用者說
                           # 裁切不好看），所以收到「剛好塞滿不裁」的最大值。
SUBJ_BOTTOM = 0.845        # 角色下緣（佔卡高）。放大時**往上長**，底邊釘住不動——
                           # 文字框上緣在 80.4%，往下長會把字蓋掉。
MASK_EDGE = 300            # foil mask 只是剪影，不必跟圖層同解析度

RARITY = 'special'


def frames(path):
    """qtrle/argb → RGBA。alpha 是直式（非預乘），實測半透明像素除以 alpha 會爆掉，
    所以**不要** unpremultiply——那是上一支黑底 mp4 才需要的補救。"""
    raw = subprocess.run(['ffmpeg', '-v', 'error', '-i', str(path),
                          '-f', 'rawvideo', '-pix_fmt', 'rgba', '-'],
                         capture_output=True, check=True).stdout
    return np.frombuffer(raw, np.uint8).reshape(-1, 720, 1280, 4)


def build_background():
    im = Image.open(BG_SRC).convert('RGB')
    # 5:7 純裁切（不重畫、不拉伸變形）：315×444 比 5:7 高一點點，裁掉多的高
    want_h = round(im.width * 7 / 5)
    if im.height > want_h:                       # 置中裁，構圖上下都不偏
        off = (im.height - want_h) // 2
        im = im.crop((0, off, im.width, off + want_h))
    return im.convert('RGBA').resize(CARD, Image.Resampling.LANCZOS), im.size


def union_box(seqs):
    """兩支動畫共用同一個裁切框：切表情時角色不能跳位或改大小。"""
    acc = None
    for seq in seqs:
        for f in seq:
            a = f[:, :, 3]
            acc = a if acc is None else np.maximum(acc, a)
    return Image.fromarray(acc).getbbox()


def place(seq, box):
    """把角色逐格合成到 600×840 的透明畫布上，位置與縮放整段固定（不能逐格重算，
    不然角色會在動畫裡自己抖動）。"""
    cw, ch = box[2] - box[0], box[3] - box[1]
    w = round(CARD[0] * SUBJ_WIDTH)
    h = round(ch * w / cw)
    x = (CARD[0] - w) // 2                       # 放大到超過卡寬時左右各裁掉一點
    y = round(CARD[1] * SUBJ_BOTTOM) - h
    out = []
    for f in seq:
        sub = Image.fromarray(f).crop(box).resize((w, h), Image.Resampling.LANCZOS)
        canvas = Image.new('RGBA', CARD, (0, 0, 0, 0))
        canvas.paste(sub, (x, y), sub)
        out.append(canvas)
    return out, {'source_bbox': list(box), 'placed': [x, y, w, h],
                 'plate_top_px': round(CARD[1] * (1 - 0.034 - 0.162)),
                 'overlaps_plate_px': max(0, y + h - round(CARD[1] * (1 - 0.034 - 0.162))),
                 'cropped_each_side_px': max(0, -x)}


def anim(images, path, quality, edge=None):
    if edge:
        size = (round(CARD[0] * edge / max(CARD)), round(CARD[1] * edge / max(CARD)))
        images = [im.resize(size, Image.Resampling.LANCZOS) for im in images]
    buf = io.BytesIO()
    images[0].save(buf, format='WEBP', save_all=True, append_images=images[1:],
                   duration=33, loop=0, quality=quality, method=4, minimize_size=True)
    path.write_bytes(buf.getvalue())
    return len(buf.getvalue())


def palette(im):
    """背景配色直接取自背景圖本身（這張卡的背景是圖不是 CSS，但卡外光暈仍吃 --pal-*）。"""
    small = im.convert('RGB').resize((48, 48), Image.Resampling.LANCZOS)
    a = np.asarray(small).reshape(-1, 3).astype(float)
    mean = a.mean(0)
    dark = np.clip(mean * 0.22, 0, 255)
    glow = np.clip(mean * 0.62, 0, 255)
    accent = np.clip(mean * 1.18, 0, 255)
    hexof = lambda v: '#%02x%02x%02x' % tuple(int(round(x)) for x in v)
    return {'base': hexof(dark), 'glow': hexof(glow), 'accent': hexof(accent),
            'ink': hexof(np.clip(mean * 0.12, 0, 255))}
