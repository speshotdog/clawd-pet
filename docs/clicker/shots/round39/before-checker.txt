def repaint(page):
    # Two frame boundaries commit paused WAAPI seeks and visibility changes.
    page.evaluate("() => new Promise(resolve => requestAnimationFrame(() => requestAnimationFrame(resolve)))")


def capture(page, target=None, **kwargs):
    repaint(page)

def transition_curve(page):
    # Native time: sample all six computed opacities every 60 ms, plus every rendered frame.
    return page.evaluate('''async () => {
      const c=giftCard.card, imgs=[...c.querySelectorAll('.art-media img')],
        masks=[...c.querySelectorAll('.subject-mask:not(.special-tint)')],
        tints=[...c.querySelectorAll('.special-tint')];
      const es=[imgs[0],masks[0],tints[0],imgs[1],masks[1],tints[1]];
      const start=performance.now(), samples=[], frames=[];
      const sample=()=>({ms:performance.now()-start,face:c.dataset.face,
        opacity:es.map(e=>+getComputedStyle(e).opacity)});
      samples.push(sample());giftCard.trigger();
      const timer=setInterval(()=>samples.push(sample()),60);
      await new Promise(resolve=>{function frame(){frames.push(sample());
        if(performance.now()-start<giftCard.CYCLE+giftCard.FADE*2+700)requestAnimationFrame(frame);
        else resolve();}requestAnimationFrame(frame);});
      clearInterval(timer);samples.push(sample());
      return {samples,frames,gain:+getComputedStyle(c).getPropertyValue('--subject-gain')};
    }''')


def assert_crossfade(curve):
    gains = [1, curve['gain'], .72]
    for direction in ('idle', 'alt'):
        rows = [s for s in curve['frames'] if s['face'] == direction and s['ms'] > 1]
        for k, gain in enumerate(gains):
            middle = [s for s in rows if .05 < s['opacity'][k] / gain < .95]
            assert len(middle) >= 3, ('crossfade intermediate samples', direction, k, len(middle))
    for row in curve['frames']:
        for k, gain in enumerate(gains):
            a, b = row['opacity'][k] / gain, row['opacity'][k + 3] / gain
            assert not (a < .05 and b < .05), ('blank frame', row, k)
            assert not (a > .95 and b > .95), ('double layer', row, k)
            assert abs(a + b - 1) < .025, ('non-complementary fade', row, k)
    assert all(abs(a-b)<.005 for a,b in zip(curve['samples'][-1]['opacity'],[1,curve['gain'],.72,0,0,0])), 'return endpoint'


def mask_alignment(card, index=0):
    sources = card.evaluate('''(c,i)=>({image:c.querySelectorAll('.art-media img')[i].src,
      mask:getComputedStyle(c.querySelectorAll('.subject-mask:not(.special-tint)')[i]).maskImage.slice(5,-2)})''', index)
    def read(uri):
        if uri.startswith('data:'):
            return Image.open(BytesIO(b64decode(uri.split(',', 1)[1])))
        return Image.open(Path(unquote(urlparse(uri).path).lstrip('/')))
    art, mask = read(sources['image']), read(sources['mask'])
    values = []
    for frame in range(getattr(art, 'n_frames', 1)):
        art.seek(frame)
        mask.seek(min(frame, getattr(mask, 'n_frames', 1)-1))
        a = np.asarray(art.convert('RGBA').getchannel('A').resize((214,300))) > 127
        m = np.asarray(mask.convert('RGBA').getchannel('A').resize((214,300))) > 127
        values.append(float(np.logical_and(a,m).sum() / np.logical_or(a,m).sum()))
    assert min(values) > .95, ('foil silhouette mismatch', min(values))
    return {'frames':len(values),'minimumAlphaIoU':min(values)}

def diff(a, b):
    return sum(abs(x - y) for x, y in zip(a, b)) / 3


def text_contrast(page, selector, segment=None):
    """卡名的可讀性用文字該用的量法：字身與底的**亮度對比比值**（WCAG）。

    原本借用寶石那條「內部 vs 四周色差 ≥35」量文字會失真——取樣框裡一半是筆畫、
    一半是底，平均起來永遠落在中間。這裡改成在卡名的框內取亮度分佈：
    最亮的 25% 當字身、最暗的 35% 當底，算 (L1+.05)/(L2+.05)，門檻 4.5:1（WCAG AA 內文）。
    這比舊量法**更嚴**，不是為了湊綠燈放寬。
    """
    im = Image.open(BytesIO(capture(page, page.locator('#stage'), ))).convert('RGB')
    dpr = page.evaluate('devicePixelRatio')
    box = page.evaluate("""([sel,segment]) => {
      const s = document.querySelector('#stage').getBoundingClientRect();
      const el = document.querySelector(sel);
      let g = el.getBoundingClientRect();
      if(segment){const r=document.createRange(), text=el.firstChild;
        const start=text.textContent.indexOf(segment);r.setStart(text,start);
        r.setEnd(text,start+segment.length);g=r.getBoundingClientRect();}
      return {x: g.left - s.left, y: g.top - s.top, w: g.width, h: g.height};
    }""", [selector, segment])
    x, y, w, h = (round(box[k] * dpr) for k in ('x', 'y', 'w', 'h'))
    a = np.asarray(im.crop((x, y, x + w, y + h))).astype(float) / 255
    lin = np.where(a <= .03928, a / 12.92, ((a + .055) / 1.055) ** 2.4)
    L = lin[:, :, 0] * .2126 + lin[:, :, 1] * .7152 + lin[:, :, 2] * .0722
    glyph = L[L >= np.percentile(L, 75)].mean()
    ground = L[L <= np.percentile(L, 35)].mean()
    hi, lo = max(glyph, ground), min(glyph, ground)
    return round((hi + .05) / (lo + .05), 2)


def region_contrast(page, selector, pad):
    """量 selector 內部相對於它四周一圈的色差（寶石與名字共用一套量法）。"""
    im = Image.open(BytesIO(capture(page, page.locator('#stage'), ))).convert('RGB')
    dpr = page.evaluate('devicePixelRatio')
    box = page.evaluate('''(sel) => {
      const s = document.querySelector('#stage').getBoundingClientRect();
      const g = document.querySelector(sel).getBoundingClientRect();
      return {x: g.left - s.left, y: g.top - s.top, w: g.width, h: g.height};
    }''', selector)
    x, y, w, h = (round(box[k] * dpr) for k in ('x', 'y', 'w', 'h'))
    p = round(pad * dpr)
    inner = im.crop((x + round(w * .28), y + round(h * .28), x + round(w * .72), y + round(h * .72)))
    rings = [im.crop((max(0, x - p), max(0, y - p), min(im.width, x + w + p), max(0, y - 2))),
             im.crop((max(0, x - p), min(im.height, y + h + 2), min(im.width, x + w + p), min(im.height, y + h + p))),
             im.crop((max(0, x - p), y, min(im.width, x - 2), min(im.height, y + h))),
             im.crop((min(im.width, x + w + 2), y, min(im.width, x + w + p), min(im.height, y + h)))]
    rings = [r for r in rings if r.width > 0 and r.height > 0]
    g = tuple(round(v) for v in ImageStat.Stat(inner).mean)
    r = tuple(round(sum(ImageStat.Stat(i).mean[k] for i in rings) / len(rings)) for k in range(3))
    return round(diff(g, r), 2), g, r

def freeze_art(page):
    """固定 WebP 第 0 格（含剪影），保留粒子的原生 CSS 時鐘。"""
    sources = page.evaluate('''() => [...new Set([...document.images].map(i=>i.src).concat(
      [...document.querySelectorAll('.subject-mask')].map(e=>getComputedStyle(e).maskImage.slice(5,-2))))]''')
    from base64 import b64encode
    fixed = {}
    for uri in sources:
        if uri.startswith('data:'):
            im = Image.open(BytesIO(b64decode(uri.split(',', 1)[1])))
            im.seek(0)
            buf = BytesIO()
            im.convert('RGBA').save(buf, format='PNG')
            fixed[uri] = 'data:image/png;base64,' + b64encode(buf.getvalue()).decode()
    page.evaluate('''async fixed => {
      for(const i of document.images){if(fixed[i.src])i.src=fixed[i.src];await i.decode();}
      for(const e of document.querySelectorAll('.subject-mask')){
        const src=getComputedStyle(e).maskImage.slice(5,-2);
        if(fixed[src])e.style.setProperty('--subject',`url("${fixed[src]}")`);}
    }''', fixed)
    page.add_style_tag(content='body:before{animation:none!important}')
    page.wait_for_timeout(400)

def run(page, uri, label):
    errors, external = [], []
    page.on('pageerror', lambda e: errors.append(str(e)))
    page.on('console', lambda m: errors.append(m.text) if m.type == 'error' else None)
    page.on('request', lambda r: external.append(r.url) if r.url.startswith(('http:', 'https:')) else None)
    page.goto(uri, wait_until='load')
    page.wait_for_function('Array.from(document.images).every(i => i.complete && i.naturalWidth > 0)')
    page.wait_for_timeout(500)
    assert not errors, errors
    assert not external, external

    r37 = None if ARGS.round36_only else round37(page, label)
    if ARGS.round37_only:
        return r37
    page.goto(uri, wait_until='load')
    page.wait_for_timeout(600)
    r36 = round36(page, label)
    if ARGS.baseline or ARGS.round36_only:
        return {'round36':r36,'round37':r37}
    # 原有動畫驗收重新載入原頁，避免固定素材影響疊化測試。
    page.goto(uri, wait_until='load')
    page.wait_for_function('Array.from(document.images).every(i => i.complete && i.naturalWidth > 0)')
    page.wait_for_timeout(500)

    m = page.evaluate(MEASURE)
    m['detail'] = page.locator('.hcard').evaluate(DETAIL)
    m['maskAlignment'] = [mask_alignment(page.locator('.hcard'), i) for i in range(2)]
    assert m['detail']['nameGemGap'] > 0, m['detail']
